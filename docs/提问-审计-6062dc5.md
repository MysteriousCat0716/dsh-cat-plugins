### 1) `lib/recall.js:712-745` — used 结算

```js
export function markRecallUsed(db, { sid, text }) {
  if (!sid) return 0;
  // 先结算滑出判定窗口的旧记录：最近 40 条 NULL 之外的，一律判「没用上」。
  db.prepare(
    `UPDATE recall_log SET used = 0
     WHERE sid = ? AND used IS NULL AND memory_id IS NOT NULL
       AND id NOT IN (
         SELECT id FROM recall_log
         WHERE sid = ? AND used IS NULL AND memory_id IS NOT NULL
         ORDER BY id DESC LIMIT 40
       )`,
  ).run(String(sid), String(sid));
  if (!text) return 0;
  const terms = new Set([...anchorsOf(text), ...entitiesOf(text)]);
  if (terms.size === 0) return 0;
  const rows = db
    .prepare(
      `SELECT l.id, m.anchors, m.entities
       FROM recall_log l
       JOIN memories m ON m.id = l.memory_id
       WHERE l.sid = ? AND l.used IS NULL AND l.memory_id IS NOT NULL
       ORDER BY l.id DESC
       LIMIT 40`,
    )
    .all(String(sid));
  if (rows.length === 0) return 0;
  const update = db.prepare('UPDATE recall_log SET used = 1 WHERE id = ?');
  let marked = 0;
  for (const row of rows) {
    const owned = [...parseList(row.anchors), ...parseList(row.entities)];
    if (owned.some((token) => terms.has(token))) {
      update.run(row.id);
    }
  }
  return marked;
}
```

意图：`used` 三态——`NULL` 未判、`1` 用上了、`0` 没用上。原来只有「写 1」，没有「写 0」，滑出 40 条窗口的旧行永远 NULL，`used=1 / 总注入` 的分母失真。

### 2) `lib/originals.js:104-119` — spill 路径

```js
/**
 * spill 路径：工具结果超过 token 阈值时，DSH 把全文写到
 * `%TEMP%/dsh-spill-XXXXXX/session-<hash>/<rand>-<tool>.txt`，正文只留预览和一行
 * `(N bytes omitted. Full formatted result stored at: <path>. Use read with offset/limit, ...)`。
 * 开括号是 `(` 不是 `[`；路径以 `.txt` 结尾；日志里有时脱敏成 `<path>`，匹配不到就返回空串。
 */
const SPILL_PATH = /Full formatted result stored at:\s*(\S+?\.txt)/;

export function spillPathOf(text) {
  const value = String(text ?? '');
  const match = value.match(SPILL_PATH);
  return match ? match[1].trim() : '';
}
```

意图：真实格式是 `(N bytes omitted. Full formatted result stored at: <path>. Use read with offset/limit, ...)`，路径以 `.txt` 结尾；`<path>` 是脱敏占位，不能当真实路径去读。
